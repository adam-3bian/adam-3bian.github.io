{%- comment -%} Native collection ordering shared by directories, home and reverse relationships. Explicit undated records never inherit Jekyll's fallback build date as a fact. Optional as_of is for isolated date-boundary checks; production uses site.time. {%- endcomment -%} {%- assign chronological_source = include.items | where_exp: 'item', 'item.alternate != true' | where_exp: 'item', 'item.redirect_to == nil' -%} {%- assign chronological_dated = chronological_source | where_exp: 'item', 'item.undated != true' | sort: 'date' -%} {%- assign chronological_undated = chronological_source | where: 'undated', true | sort_natural: 'title' -%} {%- if include.kind == 'news' -%} {%- assign chronological_items = chronological_dated | reverse | concat: chronological_undated -%} {%- elsif include.status == 'past' or include.status == 'upcoming' -%} {%- assign chronological_items = '' | split: ',' -%} {%- assign event_today = include.as_of | default: site.time | date: '%Y-%m-%d' -%} {%- for event in chronological_dated -%} {%- assign event_end = event['end-date'] | default: event.date | date: '%Y-%m-%d' -%} {%- if include.status == 'past' and event_end < event_today -%} {%- assign chronological_items = chronological_items | push: event -%} {%- elsif include.status == 'upcoming' and event_end >= event_today -%} {%- assign chronological_items = chronological_items | push: event -%} {%- endif -%} {%- endfor -%} {%- else -%} {%- assign chronological_items = chronological_dated | concat: chronological_undated -%} {%- endif -%}
